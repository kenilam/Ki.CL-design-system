@layer components, utilities;

@layer components {
  /* Two classes, but in the earlier layer */
  .kicl--components--card.kicl--components--card--size--sm {
    padding: var(--kicl--components--card--spacing);
  }
}

@layer utilities {
  /* One class, in the later layer, so it wins */
  .kicl-padding-wide {
    padding: var(--kicl-gutter-wide);
  }
}
